number: 14
query: |
  SELECT SUM(od."Quantity") AS "Quantity of Beverage bottles" 
  FROM order_details AS od, 
      products AS p, 
      categories AS c 
  WHERE od."ProductID" = p."ProductID" 
    AND p."CategoryID" = c."CategoryID" 
    AND c."CategoryName" = 'Beverages';
---
