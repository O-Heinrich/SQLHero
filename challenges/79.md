query: |
  SELECT s.name "Schüler", COALESCE(a.name, 'keine') "AG" FROM schueler s 
  LEFT JOIN (
      teilnahme t JOIN ag a ON t.ag_id = a.id
  ) ON s.id = t.schueler_id
  ORDER BY 2;
