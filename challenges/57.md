number: 57
query: |
  SELECT 
    s.name 
  FROM 
    schueler s 
  JOIN 
    klasse k ON s.klasse_id = k.id 
  JOIN 
    unterricht u ON k.id = u.klasse_id 
  JOIN 
    raum r ON u.raum_id = r.id 
  WHERE 
    r.nummer = 'R112';
---
