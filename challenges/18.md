schema: "/databases/nordwind.sql"

difficulty: "medium"
number: 18
query: |
  SELECT p."ProductName",  
       SUM(od."UnitPrice" * od."Quantity") AS "Umsatz"  
  FROM products AS p,  
      order_details AS od  
  WHERE p."ProductID" = od."ProductID"  
  GROUP BY 1;
---
