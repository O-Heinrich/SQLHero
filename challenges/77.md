query: |
  SELECT DISTINCT l1.name FROM lehrer l1 
  JOIN unterricht u1 ON l1.id = u1.lehrer_id
  JOIN unterricht u2 ON u1.fach = u2.fach
  JOIN lehrer l2 ON u2.lehrer_id = l2.id
  WHERE l2.name = 'Zimmermann' AND l2.name != l1.name;
