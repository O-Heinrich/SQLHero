query: |
  SELECT ku."CompanyName",  
       COUNT(be."CustomerID") AS "Bestellanzahl"  
  FROM orders AS be,  
      customers AS ku  
  WHERE ku."CustomerID" = be."CustomerID"  
    AND be."EmployeeID" BETWEEN 7 AND 9  
  GROUP BY ku."CompanyName"  
  HAVING COUNT(be."CustomerID") > 5; 
---

## Bestimmte Kunden 2
