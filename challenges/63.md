query: |
  SELECT l.name, l.vorname 
  FROM lehrer l 
  JOIN unterricht u ON l.id = u.lehrer_id 
  JOIN klasse k ON u.klasse_id = k.id 
  JOIN schueler s ON k.id = s.klasse_id 
  WHERE s.name = 'Schmidt' 
  UNION SELECT l.name, l.vorname 
  FROM lehrer l 
  JOIN klasse k ON l.id = k.klassenlehrer_id 
  JOIN schueler s ON k.id = s.klasse_id 
  WHERE s.name = 'Schmidt'
  ORDER BY name;

## Disziplinarkonferenz
