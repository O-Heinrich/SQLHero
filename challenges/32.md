query: |
  SELECT l."SupplierID" AS "Lieferantennummer",  
       k."CategoryID" AS "Kategoriennummer",  
       SUM(a."UnitsInStock") AS "Summe Lagerbestand (sortiert)"
  FROM suppliers AS l  
  JOIN products AS a ON l."SupplierID" = a."SupplierID"  
  JOIN categories AS k ON a."CategoryID" = k."CategoryID"  
  GROUP BY l."SupplierID", k."CategoryID"  
  HAVING SUM(a."UnitsInStock") > 50  
    AND SUM(a."UnitsInStock") < 100
  ORDER BY SUM(a."UnitsInStock");
