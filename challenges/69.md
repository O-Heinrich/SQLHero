query: |
  SELECT s2.name, s2.vorname FROM schueler s1 
  JOIN schueler s2 ON s1.klasse_id = s2.klasse_id 
  WHERE s1.name = 'Ebert' AND s1.vorname = 'Anne' AND s2.id != s1.id;
