schema: "/databases/nordwind.sql"

number: 29
query: |
  SELECT p."LastName",  
       SUM((bd."UnitPrice" * bd."Quantity") * (1 - bd."Discount")) AS "Gesamtumsatz"  
  FROM employees AS p,  
      order_details AS bd,  
      orders AS b  
  WHERE p."EmployeeID" = b."EmployeeID"  
    AND b."OrderID" = bd."OrderID"  
  GROUP BY p."LastName"  
  ORDER BY "Gesamtumsatz" ASC;  
---
