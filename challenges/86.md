query: |
  SELECT
    to_char(pa."PatAuf_AufnahmeDatum", 'DD.MM.YYYY') AS "Aufnahmedatum",
    to_char(pa."PatAuf_EntlassDatum", 'DD.MM.YYYY') AS "Entlassdatum",
    (pa."PatAuf_EntlassDatum" - pa."PatAuf_AufnahmeDatum") AS "Dauer",
    z."Z_ZimmerNummer" AS "Zimmernummer",
    s."Stat_Station" AS "Station",
    b."Bett_Nummer"
  FROM patient_aufenthalt AS pa
  LEFT JOIN zimmer AS z ON pa."PatAuf_ZID" = z."Z_ID"
  LEFT JOIN bett AS b ON z."Z_BettID" = b."Bett_ID"
  LEFT JOIN station AS s ON z."Z_StatID" = s."Stat_ID"
  WHERE
    pa."PatAuf_AufnahmeDatum" < '2020-03-01'
    AND pa."PatAuf_EntlassDatum" >= '2020-02-01'
  ORDER BY pa."PatAuf_ID";
