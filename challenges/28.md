query: |
  SELECT CONCAT(p."FirstName", ' ', p."LastName"),  
       COUNT(*) AS "Anzahl Bestellungen aus Deutschland",  
       kd."CompanyName"  
  FROM employees AS p,  
      orders AS b,  
      order_details AS bd,  
      products AS a,  
      customers AS kd  
  WHERE p."EmployeeID" = b."EmployeeID"  
    AND b."OrderID" = bd."OrderID"  
    AND bd."ProductID" = a."ProductID"  
    AND b."CustomerID" = kd."CustomerID"  
    AND kd."Country" LIKE 'Deutschland'  
    AND a."ProductName" LIKE 'Lakkalikööri'  
  GROUP BY p."LastName", p."FirstName", kd."CompanyName"
  HAVING COUNT(*) >= 2;
