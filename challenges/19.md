---
title: "Umsatz pro Auslaufartikel"
schema: "/databases/nordwind.sql"
pdf: "/databases/pdf/NordwindDB.pdf"
number: 19
query: |
  SELECT p."ProductName",  
       ROUND(SUM(od."Quantity" * od."UnitPrice")) AS "Umsatz"  
  FROM products AS p,  
      order_details AS od  
  WHERE p."ProductID" = od."ProductID"  
    AND p."Discontinued" = TRUE  
  GROUP BY p."ProductName"  
  ORDER BY 2 ASC;
---
