---
title: "Bestellungen pro Mitarbeiter"
schema: "/databases/nordwind.sql"
pdf: "/databases/pdf/NordwindDB.pdf"
difficulty: "hard"
number: 17
query: |
  SELECT e."EmployeeID", 
       e."LastName", 
       e."FirstName", 
       COUNT(o."OrderID") AS "Anzahl Bestellungen"
  FROM employees AS e
  LEFT JOIN orders AS o ON e."EmployeeID" = o."EmployeeID"
  GROUP BY e."EmployeeID", e."LastName", e."FirstName"
  ORDER BY "Anzahl Bestellungen" DESC;
---
