  SELECT 
    p.P_Bezeichnung AS Partei, 
    (
        SELECT ROUND(AVG(ww.W_Alter), 0) FROM waehler ww 
        WHERE ww.W_Geschlecht LIKE 'w' AND ww.W_P_ID = p.P_ID
    ) AS W_Alter, 
    (
        SELECT CAST(AVG(wm.W_Alter) AS INTEGER) FROM waehler wm 
        WHERE wm.W_Geschlecht LIKE 'm' AND wm.W_P_ID = p.P_ID
    ) AS M_Alter 
  FROM partei p;
