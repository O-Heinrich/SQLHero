query: |
  SELECT 
    suppliers."CompanyName", 
    products."ProductName" 
  FROM 
    products, 
    suppliers 
  WHERE 
    suppliers."SupplierID" = products."SupplierID" 
  AND 
    suppliers."CompanyName" = 'Norske Meierier' 
  ORDER BY 
    "ProductName";
