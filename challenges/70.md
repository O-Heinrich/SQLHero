query: |
  SELECT l2.name, l2.vorname FROM lehrer l2 JOIN unterricht u ON l2.id = u.lehrer_id 
  JOIN klasse k ON u.klasse_id = k.id 
  JOIN lehrer l1 ON k.klassenlehrer_id = l1.id 
  WHERE l1.name = 'Buttenmüller';

## Fachlehrer
